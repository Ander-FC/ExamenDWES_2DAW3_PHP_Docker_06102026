CREATE DATABASE CAE;

USE CAE;

CREATE TABLE SOLICITUD (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50),
    apellidos VARCHAR(100),
    dni VARCHAR(9),
    f_nac DATE,
    tlf VARCHAR(15),
    email VARCHAR(100),
    profesion VARCHAR(30),
    jornadaParcial BOOLEAN,
    idiomas VARCHAR(50)
);
